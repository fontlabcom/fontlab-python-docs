    fontlab.YPanelWidget

    YPanelWidget

    PythonQtInstanceWrapper object

    Custom dockable panel.

    What is YPanelWidget?

    YPanelWidget is the content class for a dockable panel. Subclass it to create a panel that appears in FontLab's sidebar. You are responsible for setting the title, icon, and minimum size, and for building the Qt widget hierarchy that populates the panel.

    Tutorial

    import fontlab as fl
from fontlab import YPanelManager, YPanelWidget

# ============================================================
# Part 1: Inspect the existing panels.
#
# This is useful for understanding which panels are currently
# registered and visible. Run this in the scripting panel.
# ============================================================

pm = YPanelManager.instance()

print(f"Registered panels: {len(pm.registeredPanels)}")
for panel in pm.registeredPanels:
    is_visible = pm.panelVisible(panel)
    status = "open" if is_visible else "closed"
    print(f"  [{status}] id={panel.id}  title='{panel.title}'")

print()
print(f"Currently open: {len(pm.panels)}")

    Common Scripting Patterns

    Collapse and expand a panel programmatically
    from fontlab import YPanelManager

pm    = YPanelManager.instance()
panel = next((p for p in pm.registeredPanels if 'Layers' in p.title), None)
if panel:
    panel.collapsed = not panel.collapsed   # toggle
    print(f"'{panel.title}' collapsed: {panel.collapsed}")
    Check minimum panel size before resizing
    from fontlab import YPanelManager

pm = YPanelManager.instance()
for panel in pm.panels:
    min_size = panel.minimumPanelSize
    print(f"'{panel.title}': min {min_size.width()}×{min_size.height()}")
