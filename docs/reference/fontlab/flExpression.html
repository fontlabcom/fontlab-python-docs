    fontlab.flExpression

    flExpression

    Formula string for axis-driven values.

    What is flExpression?

    An flExpression stores and evaluates a text formula that can reference design-space axis values and font metric names. For example:

    "ascender * 0.7"
"xHeight + stemWidth"
"weight < 400 ? 50 : 100"

    When FontLab evaluates an expression, it substitutes the current values of axis tags (like wght, wdth) and metric names (like ascender, xHeight, capHeight) for the variable names in the formula, then computes the result.

    Expressions are used by guidelines, anchors, and parametric shape nodes to make their positions vary smoothly across the design space.

    Tutorial

    from PythonQt.QtCore import QPointF
import fontlab as fl
from fontlab import flExpression

# ============================================================
# 1. Create an expression and inspect it.
#    This is standalone — not yet attached to any object.
# ============================================================

expr = flExpression()
expr.expression = "ascender * 0.7"

print("Has expression:", expr.hasExpression())  # True
print("Has error:", expr.hasError())             # False (not yet evaluated)
print("Is ready:", expr.ready)                   # False (not yet evaluated)

# ============================================================
# 2. Inspect which variables the formula depends on.
# ============================================================

deps = expr.variables({})   # {} = no alias substitution
print("Depends on:", deps)  # ['ascender']

# ============================================================
# 3. Rename variables for human-friendly display.
#    Useful if you are showing the expression in a UI panel.
# ============================================================

alias_map = {"ascender": "Ascender Height"}
friendly = expr.expression(alias_map)
print("Friendly formula:", friendly)  # "Ascender Height * 0.7"

# ============================================================
# 4. Read an expression from an existing guideline.
#    Guidelines can be formula-driven — for example, the
#    x-height guideline might use "xHeight" as its expression.
# ============================================================

font = fl.CurrentFont()
for gl in font.guidelines:
    # flGuideLine exposes its expression through the y-property:
    e = gl.expressionY  # returns flExpression
    if e and e.hasExpression():
        print(f"Guideline at y={gl.centerPoint.y():.0f}: "
              f"formula = '{e.expression}'")
        if e.hasError():
            from fontlab import flPropertiesProcessor
            err_msg = flPropertiesProcessor.errorToStr(e.error)
            print(f"  ERROR: {err_msg}")

# ============================================================
# 5. Assign an expression to a guideline.
#    Make a horizontal guideline track the x-height
#    so it always sits at the right height in every master.
# ============================================================

from fontlab import flGuideLine

# Create a new guideline at y=500 initially:
gl = flGuideLine(0.0, QPointF(0, 500))
font.appendGuideline(gl)

# Now attach an expression so it tracks xHeight automatically:
gl.expressionY.expression = "xHeight"
gl.expressionY.resetReady()   # force re-evaluation

fl.Update()
print("Guideline is now tracking xHeight.")

    Common Scripting Patterns

    Validate a formula before assigning it
    import fontlab as fl
from fontlab import flExpression

font = fl.CurrentFont()
# Test expressions before applying:
for formula in ["=xHeight", "=capHeight * 0.5", "=badVariable + 100"]:
    expr = flExpression()
    expr.expression = formula
    valid = expr.validate(font)   # check against font's variable scope
    print(f"'{formula}': valid={valid}")
    Copy a formula from one guideline to another
    import fontlab as fl

font = fl.CurrentFont()
guides = font.guidelines

if len(guides) >= 2:
    src_formula = guides[0].expression or ""
    if src_formula:
        guides[1].expression = src_formula
        fl.Update()
        print(f"Copied formula '{src_formula}' to '{guides[1].name}'")
