    fontlab.flMarkerPoint

    flMarkerPoint

    Snap Line Measurement Marker.

    What is flMarkerPoint?

    Guidelines in FontLab can carry measurement markers — small tick marks that represent stem widths, zone boundaries, or sidebearing measurements. An flMarkerPoint is one such tick mark. It sits at a specific position along the guideline and carries flag bits that describe what kind of measurement it represents.

    Most designers interact with markers through the Snap/Measure panel rather than in scripts. But if you are writing a tool that programmatically creates or analyses measurement data attached to guidelines, flMarkerPoint gives you direct access to each marker's position and role flags.

    Tutorial

    import fontlab as fl

# ============================================================
# Inspect all marker points on all guidelines in the current
# glyph layer.  Each guideline can carry zero or more markers.
# ============================================================

font  = fl.CurrentFont()
glyph = font['H']   # 'H' often has stems worth measuring
layer = glyph.layers[0]

for guide in layer.guides:
    markers = guide.markers   # list of flMarkerPoint
    if not markers:
        continue
    print(f"Guideline at angle {guide.angle:.0f}°:")
    for m in markers:
        kind = "stem" if m.isStem else "other"
        sel  = " [selected]" if m.isSelected else ""
        print(f"  position={m.position:.0f}  type={kind}  "
              f"key={m.key}  flags={m.markerFlags:#010b}{sel}")

# ============================================================
# Create a new stem marker and attach it to the first guide.
# ============================================================

if layer.guides:
    guide = layer.guides[0]

    new_marker = fl.flMarkerPoint.createInstance()
    new_marker.position = 80.0   # 80 units from guideline centre
    new_marker.isStem   = True   # mark it as a stem edge

    guide.addMarker(new_marker)
    fl.Update()
    print(f"\nAdded stem marker at position 80 to first guideline.")

    Common Scripting Patterns

    List all hand-placed markers across all glyphs
    import fontlab as fl

font = fl.CurrentFont()

for glyph in font.glyphs:
    layer = glyph.layers[0]
    for guide in layer.guides:
        for m in guide.markers:
            if str(m.key) == 'HandMarker':
                print(f"{glyph.name}: hand marker at {m.position:.0f}")
    Check if a guideline has any stem markers
    import fontlab as fl

font  = fl.CurrentFont()
glyph = font['H']
layer = glyph.layers[0]

for i, guide in enumerate(layer.guides):
    stem_markers = [m for m in guide.markers if m.isStem]
    print(f"Guide {i}: {len(stem_markers)} stem marker(s)")
