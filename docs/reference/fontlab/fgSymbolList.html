    fontlab.fgSymbolList

    fgSymbolList

    A string of characters.

    What is fgSymbolList?

    An fgSymbolList is an ordered, mutable sequence of fgSymbol objects — essentially, a text string at the font object level. Where Python's str stores Unicode code points and knows nothing about your font, fgSymbolList stores symbols that are already resolved to specific glyphs in a specific font.

    You use fgSymbolList when you need to: - Programmatically set the text string in a metrics view text block - Inspect which characters are selected in a metrics view - Build a sequence of glyphs for batch measurement or spacing work

    Tutorial

    import fontlab as fl
from fontlab import fgSymbol, fgSymbolList

# ============================================================
# Build a symbol list from a Unicode string — the easiest way.
# This creates one symbol per character in the string.
# ============================================================

sl = fgSymbolList()
sl.setString("HAMBURGEFONS")   # a classic font-testing string
print(f"String: '{sl.string(False)}'  ({sl.size()} chars)")

# ============================================================
# Iterate symbols and print each character's name and unicode.
# ============================================================

for i in range(sl.size()):
    s = sl.at(i)
    print(f"  [{i}] U+{s.unicode:04X}  '{s.name}'")

# ============================================================
# Build a list from Unicode code points directly.
# Useful when you have a list of codepoints from another source.
# ============================================================

sl2 = fgSymbolList()
sl2.setUnicodeList([0x0041, 0x0042, 0x0043])   # A, B, C
print(f"\nFrom unicodes: '{sl2.string(False)}'")

# ============================================================
# Build a list from glyph names.
# Useful when working with non-unicode glyphs like ligatures.
# ============================================================

sl3 = fgSymbolList()
sl3.setNamesList("H A M B U R G")
print(f"From names: '{sl3.string(False)}'  ({sl3.size()} symbols)")

# ============================================================
# Extract a sub-string — like Python's slicing.
# ============================================================

sub = sl.mid(0, 7)     # first 7 characters: "HAMBURG"
print(f"\nFirst 7: '{sub.string(False)}'")

# ============================================================
# Append one list to another.
# ============================================================

tail = fgSymbolList()
tail.setString(" test")
sl.append(tail)
print(f"After append: '{sl.string(False)}'  ({sl.size()} chars)")

# ============================================================
# Compare two lists for equality.
# ============================================================

sl_a = fgSymbolList()
sl_a.setString("ABC")
sl_b = fgSymbolList()
sl_b.setString("ABC")
sl_c = fgSymbolList()
sl_c.setString("ABD")

print(f"\nABC == ABC: {sl_a.equals(sl_b)}")  # True
print(f"ABC == ABD: {sl_a.equals(sl_c)}")  # False

# ============================================================
# Remove a range of symbols.
# ============================================================

sl4 = fgSymbolList()
sl4.setString("Hello World")
sl4.remove(5, 6)   # remove " World" (index 5, count 6)
print(f"After remove: '{sl4.string(False)}'")  # "Hello"

# ============================================================
# Use symbols() to get a plain Python list for iteration.
# ============================================================

all_syms = sl_a.symbols()   # returns a Python list
for sym in all_syms:
    print(f"  {sym.name}")

    Common Scripting Patterns

    Build a metrics-view text string from a font's glyph names
    from fontlab import fgSymbolList

sl = fgSymbolList()
sl.setNamesList("H O spacing A B C D E F G")
print(sl.string(False))
    Get the selected characters from the metrics view
    import fontlab as fl

ws     = fl.flWorkspace.instance()
canvas = ws.getActiveCanvas()
if canvas:
    tb = canvas.textBlock() if hasattr(canvas, 'textBlock') else None
    if tb:
        selected = tb.getSelectedCharacters()
        if selected:
            print("Selected:", selected.string(False))
    Truncate a long string to fit a line
    from fontlab import fgSymbolList

sl = fgSymbolList()
sl.setString("A very long string that might not fit in a single line of the metrics view")
truncated = sl.limitLineLength(20)
if truncated:
    print("String was truncated to 20 chars")
print(sl.string(False))
