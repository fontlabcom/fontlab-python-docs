    fontlab.flSticker

    flSticker

    Canvas Annotation.

    What is flSticker?

    A sticker is a text label that you can attach to a shape in the glyph editor. It appears directly on the canvas as a coloured text box, floating next to the shape it belongs to. Stickers are a handy way to annotate specific shapes during a review or proofing process — you can leave notes like "check spacing", "needs smoother curve", or "discuss with client" directly on the glyph.

    In scripts, flSticker lets you add, read, move, or remove these canvas annotations programmatically. For example, you could write a script that audits all glyphs and attaches stickers to flag potential problems.

    Tutorial

    from PythonQt.QtCore import QPointF
import fontlab as fl
from fontlab import flSticker

# ============================================================
# Add an annotation sticker to a shape in the current glyph.
# This is useful for leaving review comments directly on
# the canvas, visible to anyone editing the glyph.
# ============================================================

font  = fl.CurrentFont()
glyph = font['A']
layer = glyph.layers[0]

if not layer.shapes:
    print("No shapes in 'A'.")
else:
    shape = layer.shapes[0]

    # Create the sticker and set its text:
    sticker = flSticker()
    sticker.text = "Check the apex"

    # Position it near the top of the shape's bounding box:
    bbox = shape.boundingBox
    sticker.move(QPointF(bbox.center().x(), bbox.top() + 20))

    # Attach to the shape:
    shape.addSticker(sticker)
    fl.Update()

    print(f"Added sticker: '{sticker.text}'")
    print(f"Position: {sticker.textRect}")

# ============================================================
# Read back all stickers on all shapes in the glyph.
# Useful for a review script that collects all annotations.
# ============================================================

print("\nAll stickers in 'A':")
for shape in layer.shapes:
    for s in shape.stickers:
        print(f"  Shape '{shape.name}': '{s.text}' at {s.textRect}")

# ============================================================
# Remove all stickers from a shape.
# Use this after a review pass to clean up annotations.
# ============================================================

for shape in layer.shapes:
    shape.clearStickers()

fl.Update()
print("All stickers cleared.")

    Common Scripting Patterns

    Flag all glyphs with open contours
    from PythonQt.QtCore import QPointF
import fontlab as fl
from fontlab import flSticker

font = fl.CurrentFont()

for glyph in font.glyphs:
    layer = glyph.layers[0]
    for shape in layer.shapes:
        for contour in shape.contours:
            if not contour.closed:
                s = flSticker()
                s.text = "OPEN CONTOUR"
                bbox = shape.boundingBox
                s.move(QPointF(bbox.left(), bbox.top() + 30))
                shape.addSticker(s)

fl.Update()
print("Flagged all open contours with stickers.")
    Collect all sticker texts across the entire font
    import fontlab as fl

font = fl.CurrentFont()
notes = []

for glyph in font.glyphs:
    for layer in glyph.layers:
        for shape in layer.shapes:
            for sticker in shape.stickers:
                notes.append((glyph.name, layer.name, sticker.text))

print(f"Found {len(notes)} stickers:")
for glyph_name, layer_name, text in notes:
    print(f"  {glyph_name} / {layer_name}: {text}")
