    fontlab.flGuideLine

    flGuideLine

    What is a guideline?

    Guidelines are the non-printing reference lines you draw across your canvas to help align things. They are the blue (or custom-colored) lines that mark cap height, x-height, descender, optical margins, stem axes, and any other alignment target you care about.

    FontLab has two scopes of guidelines:

    • Font-level guidelines live in flPackage.guidelines. They appear in every glyph you open — they are global to the whole font. These are perfect for metrics references like ascender, cap height, x-height, descender, and baseline, because those values are relevant to every glyph.

    • Glyph-level guidelines live in flLayer.guidelines. They only appear when that specific glyph (and layer) is open in the editor. Use these for glyph-specific construction lines, like the exact center of a capital O, or the shoulder angle on a lowercase n.

    Beyond simple horizontal and vertical lines, guidelines can be:

    • Angled — any angle in degrees. Useful for italic slant guides or diagonal construction lines.
    • Expression-based — a guideline whose position is tied to a font metric by a formula, like "=xHeight". If you change xHeight, the guideline moves with it automatically.
    • Power guides — guidelines that FontLab auto-generates from the extremes of your outlines. They snap to the tops, bottoms, and sides of your contours, making it easy to align multiple glyphs to common optical heights.
    What is flSnapLine?

    flSnapLine is the technical base class that handles all the snap and measurement geometry — the center node, the vector node, the snap flags, and the distance/offset math. You rarely need to interact with flSnapLine directly; flGuideLine inherits everything from it and adds the extra features (color, tags, style, expressions) that you'll actually use. Think of flSnapLine as the engine and flGuideLine as the car.

    Tutorial

    from PythonQt.QtCore import QPointF
from PythonQt.QtGui import QColor
import fontlab as fl
from fontlab import flGuideLine

font  = fl.CurrentFont()
glyph = font['H']
layer = glyph.layers[0]

# ============================================================
# 1. Font-level (global) guidelines
# ============================================================
# Font-level guidelines appear in every glyph. They're ideal for
# shared metrics references like cap height, ascender, x-height,
# and descender. Add them once and they show up everywhere.

# Create a horizontal guideline at y = 700 (cap height reference).
# angle=0 means horizontal; the QPointF gives the position.
h_guide = flGuideLine(0.0, QPointF(0, 700))
font.addGuideline(h_guide)
print("Font-level guidelines:", len(font.guidelines))

# Create a vertical guideline at x = 300 (a stem axis reference).
# angle=90 means vertical.
v_guide = flGuideLine(90.0, QPointF(300, 0))
font.addGuideline(v_guide)

# ============================================================
# 2. Glyph-level guidelines
# ============================================================
# Glyph-level guidelines only appear in the one glyph where you
# add them. Great for construction lines specific to that letter —
# like the shoulder angle on a lowercase 'n', or the center of a
# round letter's bowl.

# A 45° diagonal guideline through the point (100, 100):
a_guide = flGuideLine(45.0, QPointF(100, 100))
layer.appendGuidelines([a_guide])
print("Glyph-level guidelines in this layer:", len(layer.guidelines))

# ============================================================
# 3. Inspecting guidelines
# ============================================================
# Loop through all font-level guides and print their properties.

for gl in font.guidelines:
    print(f"  angle={gl.angle:.1f}°  "
          f"pos={gl.centerPoint}  "
          f"vertical={gl.isVertical()}  "
          f"horizontal={gl.isHorizontal()}")

# ============================================================
# 4. Moving a guideline with a transform
# ============================================================
# The guideline doesn't have a simple ".y = 720" setter, so you
# move it with applyTransform(). QTransform().translate(dx, dy)
# shifts it by (dx, dy) in font units.

gl = font.guidelines[0]
print("Before move:", gl.centerPoint)

from PythonQt.QtGui import QTransform
gl.applyTransform(QTransform().translate(0, 20))   # move up 20 units
print("After move:", gl.centerPoint)

# ============================================================
# 5. Coloring guidelines
# ============================================================
# You can assign any color to a guideline to help distinguish it.
# Red for construction guides, blue for metrics, green for
# opticals — whatever helps your visual system.

gl.color = QColor(220, 60, 60)   # a warm red
print("Has custom color:", gl.hasColor)   # True

# ============================================================
# 6. Expression-based guidelines
# ============================================================
# An expression ties the guideline's position to a font metric.
# When you update the metric (e.g. change xHeight in Font Info),
# the guideline follows automatically. The expression string starts
# with "=" followed by a metric name or arithmetic formula.

gl.expression = "=xHeight"
print("Expression:", gl.expression)
print("Position controlled by expression:", gl.hasExpression)

# You can also use arithmetic:
h_guide.expression = "=capHeight * 0.5"   # midpoint of cap height

# ============================================================
# 7. Tagging guidelines for filtering
# ============================================================
# Tags let you group guides by purpose so your scripts can
# work on just a subset without touching everything.

h_guide.tags = {"metrics", "horizontal"}
v_guide.tags = {"construction"}

# Find all guides tagged "metrics":
metric_guides = [g for g in font.guidelines if "metrics" in g.tags]
print("Metric guides:", len(metric_guides))

# ============================================================
# 8. Snap geometry (distanceTo, offsetTo)
# ============================================================
# You can query how far a point is from a guideline, or get the
# snap vector that would move a point onto the guide.
# This is useful if you want to measure how far a node is
# from an alignment target.

pt = QPointF(100, 680)
dist   = gl.distanceTo(pt, 0)    # perpendicular distance, signed
offset = gl.offsetTo(pt, 0)      # vector to snap onto the guide
print(f"Distance from point to guide: {dist:.1f} units")
print(f"Snap offset: ({offset.x():.1f}, {offset.y():.1f})")

# ============================================================
# 9. Markers on a guideline
# ============================================================
# Some guidelines carry measurement markers — tick marks that
# label distances or stem widths. If you're scripting metrics
# or stem analysis, you can read them here.

for m in gl.markers:
    print(f"  Marker at position={m.position}")

# ============================================================
# 10. Power guides (auto-generated from outlines)
# ============================================================
# Power guides are automatically created from the extreme points
# of your contours. They snap to the optical tops, bottoms, and
# sides of your shapes — useful for aligning multiple glyphs
# to a shared optical height.
#
# flGuideLine.createPowerGuides(font, layer,
#     horizontal=True,   ← capture top/bottom extremes
#     vertical=True,     ← capture left/right extremes
#     diagonal=False)    ← skip diagonal tangent points

ok = flGuideLine.createPowerGuides(font, layer,
    True, True, False,
    use_glyph_guides=True,
    use_font_guides=True)
print("Power guides created:", ok)
print("Layer guides after power-guide generation:", len(layer.guidelines))

# ============================================================
# 11. Sorting guidelines
# ============================================================
# Sort font guidelines from bottom to top (ascending y position):
sorted_guides = flGuideLine.sortGuideLines(list(font.guidelines), True)
for g in sorted_guides:
    print(f"  y={g.centerPoint.y():.0f}  angle={g.angle:.0f}°")

# ============================================================
# 12. Removing a guideline
# ============================================================

font.removeGuideline(v_guide)
print("Font guidelines remaining:", len(font.guidelines))

    Common Scripting Patterns

    Set font metrics as expression guidelines
    from PythonQt.QtCore import QPointF
# Create guides for the four main font metrics, tied to expressions
# so they stay correct even if you later change the values.
import fontlab as fl
from fontlab import flGuideLine

font = fl.CurrentFont()

metrics = [
    ("=ascender",   "metrics"),
    ("=capHeight",  "metrics"),
    ("=xHeight",    "metrics"),
    ("=descender",  "metrics"),
]

for expr, tag in metrics:
    g = flGuideLine(0.0, QPointF(0, 0))   # position is set by expression
    g.expression = expr
    g.tags = {tag}
    font.addGuideline(g)

fl.Update()
    Remove all guidelines from a layer
    # Clear every glyph-level guide from a specific layer.
# Font-level guides are untouched.
layer = fl.CurrentFont()['n'].layers[0]
layer.removeGuides(True, True)   # True, True = remove all guidelines (both orientations)
fl.Update()
    Color-code guidelines by type
    from PythonQt.QtGui import QColor
# Walk through all font guidelines and assign colors
# based on whether they're horizontal, vertical, or diagonal.
font = fl.CurrentFont()
for g in font.guidelines:
    if g.isHorizontal():
        g.color = QColor(50, 100, 200)    # blue for horizontal
    elif g.isVertical():
        g.color = QColor(200, 100, 50)    # orange for vertical
    else:
        g.color = QColor(100, 180, 100)   # green for diagonal
fl.Update()
    Measure how far nodes are from a guide
    from PythonQt.QtCore import QPointF
# Find all nodes that are within 5 units of the cap-height guide.
font = fl.CurrentFont()
cap_guide = next(
    (g for g in font.guidelines if g.expression == "=capHeight"), None
)
if cap_guide:
    glyph = font['H']
    layer = glyph.layers[0]
    for contour in layer.getContours():
        for node in contour.nodes():
            pt = QPointF(node.x, node.y)
            if abs(cap_guide.distanceTo(pt, 0)) < 5:
                print(f"Node at ({node.x}, {node.y}) is near cap height")
    Generate power guides for every glyph in the font
    # Build power guides for every glyph's first layer.
# Useful for identifying optical height variation across glyphs.
font = fl.CurrentFont()
for glyph in font.glyphs:
    layer = glyph.layers[0]
    flGuideLine.createPowerGuides(font, layer, True, True, False)
fl.Update()
